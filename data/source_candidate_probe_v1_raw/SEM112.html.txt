(function () {
  const MIN_FONT_SIZE = 24;   // px
  const MAX_FONT_SIZE = 260;  // px
  const WIDTH_PERCENT = 85;   // % of viewport

  function fitText(element) {
    if (!element) return;

    const maxWidth = window.innerWidth * (WIDTH_PERCENT / 100);

    // Reset styles before measuring
    element.style.fontSize = MAX_FONT_SIZE + 'px';
    element.style.whiteSpace = 'nowrap';
    element.style.overflowWrap = 'normal';
    element.style.wordWrap = 'normal';

    const textWidth = element.scrollWidth;

    if (textWidth <= maxWidth) return;

    const scale = maxWidth / textWidth;
    const newSize = Math.max(
      MIN_FONT_SIZE,
      Math.floor(MAX_FONT_SIZE * scale)
    );

    element.style.fontSize = newSize + 'px';

    if (newSize === MIN_FONT_SIZE) {
      element.style.whiteSpace = 'normal';
      element.style.overflowWrap = 'break-word';
      element.style.wordWrap = 'break-word';
    }
  }

  const el = document.getElementById('domain');

  // Run after fonts load
  if (document.fonts?.ready) {
    document.fonts.ready.then(() => fitText(el));
  } else {
    fitText(el);
  }

  // Throttled resize
  let resizeTimer;
  window.addEventListener('resize', () => {
    clearTimeout(resizeTimer);
    resizeTimer = setTimeout(() => fitText(el), 150);
  });
  //////////////////// button hover effect /////////////////////
  const link = document.querySelector('.adbutton');
  const domain = document.getElementById('domain');

  if (!link || !domain) return;

  // Store original text-shadow
  const originalShadow = getComputedStyle(domain).textShadow;

  link.addEventListener('mouseenter', () => {
    domain.style.textShadow = 'none';
  });

  link.addEventListener('mouseleave', () => {
    domain.style.textShadow = originalShadow;
  });
})();
</script>
