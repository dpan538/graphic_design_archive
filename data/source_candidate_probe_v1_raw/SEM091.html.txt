    $(function () {
        $( "#LoginPanel_btnLogin").click(function (event) {
            event.preventDefault();

            var username = $("#LoginPanel_tbxUsername").val();
            var password = $("#LoginPanel_tbxPassword").val();
            if (username != '' && password != '') {
                $.ajax({
                    url: "https://dar.bibalex.org//en/User/ManageCollection.aspx/LoginToBAMM",
                    data: "{ 'username': '" + username + "', 'password': '" + password + "' }",
                    dataType: "json",
                    type: "POST",
                    contentType: "application/json; charset=utf-8",


                    success: function (data) {
                        if (data.d == 'success') {
                           //location.reload(true);
                            //window.location = window.location;

                            // redirect to page
                            window.location = 'https://dar.bibalex.org/landingPage/landingPage.aspx';
                        }
                        else {
                            var loginErrorMsg = data.d;
                            $(".invalidCredentials").text(loginErrorMsg);
                            $(".invalidCredentials").show();
                        }
                    },
                    error: function () {
                        $(".loginErr").show();
                    },
                    failure: function () {
                        $(".loginErr").show();
                    }
                });
            }
        });
    });
</script>

                                    $(document).ready(function () {
                                        function updateButtonContent(selectedText, selectedIcon) {
                                            if ($(window).width() <= 765) {
                                                // Small screens: Show only the icon
                                                $("#dropdownButton").html(`<img id="dropdownIcon" src="${selectedIcon}" />`);
                                            } else {
                                                // Larger screens: Show only the text
                                                $("#dropdownButton").text(selectedText);
                                            }
                                        }

                                        $(".mediaDDL").click(function (event) {
                                            event.preventDefault(); // Prevent default link behavior

                                            let selectedText = $(this).text().trim(); // Get text
                                            let selectedValue = $(this).data("value");
                                            let selectedIcon = $(this).find("img").attr("src"); // Get icon

                                            updateButtonContent(selectedText, selectedIcon);

                                            // Store the selected value in the hidden input
                                            $("#selectedValue").val(selectedValue);
                                        });

                                        // Ensure button updates properly when resizing the window
                                        $(window).resize(function () {
                                            let selectedText = $(".mediaDDL[data-value='" + $("#selectedValue").val() + "']").text().trim();
                                            let selectedIcon = $(".mediaDDL[data-value='" + $("#selectedValue").val() + "'] img").attr("src");

                                            if (selectedText && selectedIcon) {
                                                updateButtonContent(selectedText, selectedIcon);
                                            }
                                        });
                                    });

    $(function () {
        $(".ddlType").change(function () {
            ViewSelectedAdvancedSearchTab();
        });


        ViewSelectedAdvancedSearchTab();
    });


    function ViewSelectedAdvancedSearchTab() {
        var selectedType = $(".ddlType option:selected").val();
        var typesDivs = $("div[type-index]");

        typesDivs.each(function () {
            $(this).hide();
        });

        if (selectedType >= 0 && selectedType < 10) {
            //$(typesDivs[selectedType]).show();
            $($("div[type-index='" + selectedType + "']")[0]).show();
        }
    }

    function ViewSelectedAdvancedSearchTab_1(dvId) {
        var typesDivs = $("div[type-index]");

        typesDivs.each(function () {
            $(this).hide();
        });

        /*$("#" + dvId).show();*/

        $(`div[type-index|='${dvId}']`).show();

    }

    $(function () {
        $("#clearSearch").click(function () {
            // common fields
            $("#AdvancedSearch_tbxaSearchTitle").val("");
            $("#AdvancedSearch_tbxaSearchSubject").val("");
            $("#AdvancedSearch_tbxaSearchCollection").val("");

            $("#AdvancedSearch_ddlCategory").get(0).selectedIndex = 0;

            // search type dropdownlist
            $("#AdvancedSearch_ddlaSearchType").get(0).selectedIndex = 0;
            ViewSelectedAdvancedSearchTab();

            // books fields
            $("#AdvancedSearch_tbxaSearchAuthor").val("");
            $("#AdvancedSearch_tbxaSearchPublisher").val("");
            $("#AdvancedSearch_tbxaSearchCallNumber").val("");
            $("#AdvancedSearch_tbxaSearchKeyword").val("");
        });
    });
</script>
