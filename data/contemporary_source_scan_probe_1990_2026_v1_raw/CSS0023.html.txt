function getUrlParameter(sParam){
    var sPageURL = window.location.search.substring(1), sURLVariables = sPageURL.split('&'), sParameterName, i;
    for (i = 0; i < sURLVariables.length; i++) {
        sParameterName = sURLVariables[i].split('=');
        if (sParameterName[0] === sParam) {
          return sParameterName[1] === undefined ? true : decodeURIComponent(sParameterName[1]);
        }
    }
}

function ellipsis(k){
  var length = 15;
  return getTextLength(k) > length ? k.substr(0,length)+"..." : k;
}
function getTextLength(str) {
    var len = 0;
    for (var i = 0; i < str.length; i++) {
        if (escape(str.charAt(i)).length == 6) {
            len++;
        }
        len++;
    }
    return len;
}

function getTextWidth(text, font) {
    // re-use canvas object for better performance
    var canvas = getTextWidth.canvas || (getTextWidth.canvas = document.createElement("canvas"));
    var context = canvas.getContext("2d");
    context.font = font;
    var metrics = context.measureText(text);
    return metrics.width;
}

$(document).ready(function() {
  var current_slide = 0;
  var total_slides = $('.right_content_box').length;
  var display_slides = 5;
  var animation_time = 600; // 애니메이션

  // 첫 5개
  $('.right_content_box').slice(display_slides).hide();

  $('.more_btn_r.chevron_right.GR_lst_mr').click(function() {
    slideRight();
  });

  $('.more_btn_r.chevron_left.GR_lst_mr').click(function() {
    slideLeft();
  });

  function slideRight() {
    if (current_slide < total_slides - display_slides) {
      $('.right_content_box').eq(current_slide).fadeOut(animation_time);
      $('.right_content_box').eq(current_slide + display_slides).fadeIn(animation_time);
      current_slide++;
    }
  }

  function slideLeft() {
    if (current_slide > 0) {
      $('.right_content_box').eq(current_slide - 1).fadeIn(animation_time);
      $('.right_content_box').eq(current_slide + display_slides - 1).fadeOut(animation_time);
      current_slide--;
    }
  }

  // 2초마다 자동으로 오른쪽으로 슬라이드
  setInterval(slideRight, 3000 + animation_time); // 애니메이션 시간만큼 추가
});

function goTotalSearch(dsCat,detailCat)
{
  var f = document.frmTotalSearch;

  // 중복검색이 아닌 경우 실제 검색어 할당
  if (!f.dsTxt.value)
    f.dsTxt.value = f.searchValue.value;

  str = f.dsTxt;
  if (!validFieldText(str, "검색어를")){
    return ;
  }

  // 카테고리 설정
  if (dsCat)
  {
    if (f.searchValue.value==f.searchValue.defaultValue)  // 탭검색시 중복검색일때 제외
      f.dsDup.value = "Y";

    f.dsCat.value = dsCat;

    if (detailCat)
      f.detailCat.value = detailCat;
  }

  f.action = "/dsearch/searchDetail.asp";
  if (f.dsCat.value == "0" || f.dsCat.value == "") {
    f.action = "/dsearch/";
  }

  f.submit();
}

function validateEmail(email){
  var re = /^([\w-]+(?:\.[\w-]+)*)@((?:[\w-]+\.)*\w[\w-]{0,66})\.([a-z]{2,6}(?:\.[a-z]{2})?)$/i;
  return re.test(email);
}

    if(StartDate > CurrentDate){

        ret = " cyan";

    if(Difference_In_Days < 3 && Difference_In_Days >= 0){

    return ret;
}
