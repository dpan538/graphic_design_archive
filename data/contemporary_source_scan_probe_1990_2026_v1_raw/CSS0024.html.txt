let kwdFlag = false;

function goMainSearch(){
  var f = document.forms["comSchForm"];
  if($(".hmtab").hasClass("on")){
    if(f.insiteschStr.value == ""){
      alert('검색어를 입력하세요.');
      $("input[name=insiteschStr]").focus();
    }else{
      f.kwd.value = f.insiteschStr.value;
      f.submit();
    }
  }else{
    if(f.kwd.value == ""){
      alert('검색어를 입력하세요.');
      $("input[name=kwd]").focus();
    }else{
      if(!kwdFlag){
        fn_setKwdLog(f.kwd.value);
        kwdFlag = true;
      }
      f.submit();
    }
  }
  return false;
}

function fn_setKwdLog(logKwd) {
  var dataValue =  ({"logKwd":logKwd});
  $.ajax({
    url : '/NL/search/search_kwd_log.ajax',
    data : dataValue,
    type: 'POST',
    dataType : "json",
    //async : false,
    success: function(data) {
      //console.log();
    },
    error : function(xhr) {
      //alert(xhr.status);
    }
  });

}
